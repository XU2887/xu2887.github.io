        // 添加到购物车
        function addToCart(id) {
            const product = products.find(p => p.id === id);
            const existing = cart.find(item => item.id === id);
            if (existing) {
                existing.qty += 1;
            } else {
                cart.push({ ...product, qty: 1 });
            }
            updateCartUI();
            showToast(`已将「${product.name.substring(0, 12)}...」加入购物车`);
        }

        // 更新购物车视图
        function updateCartUI() {
            const badge = document.getElementById('cart-badge');
            const totalQty = cart.reduce((sum, item) => sum + item.qty, 0);
            badge.textContent = totalQty;

            const container = document.getElementById('cart-items');
            const totalEl = document.getElementById('cart-total');

            if (cart.length === 0) {
                container.innerHTML = `<div class="text-center py-16 text-gray-400 text-sm">购物车空空如也，快去选购车队同款装备吧！</div>`;
                totalEl.textContent = '¥0';
                return;
            }

            let totalPrice = 0;
            container.innerHTML = cart.map(item => {
                totalPrice += item.price * item.qty;
                return `
                    <div class="flex items-center justify-between p-3 bg-gray-50 rounded-xl border border-gray-200">
                        <div class="pr-2">
                            <div class="text-xs font-bold text-gray-900">${item.name}</div>
                            <div class="text-xs text-yp-darkGreen font-black mt-1">¥${item.price} × ${item.qty}</div>
                        </div>
                        <div class="flex items-center space-x-2 shrink-0">
                            <button onclick="changeQty(${item.id}, -1)" class="w-6 h-6 rounded bg-gray-200 text-gray-700 font-bold text-xs">-</button>
                            <span class="text-xs font-bold">${item.qty}</span>
                            <button onclick="changeQty(${item.id}, 1)" class="w-6 h-6 rounded bg-gray-200 text-gray-700 font-bold text-xs">+</button>
                        </div>
                    </div>
                `;
            }).join('');
            totalEl.textContent = `¥${totalPrice.toLocaleString()}`;
        }

        function changeQty(id, delta) {
            const item = cart.find(i => i.id === id);
            if (!item) return;
            item.qty += delta;
            if (item.qty <= 0) {
                cart = cart.filter(i => i.id !== id);
            }
            updateCartUI();
        }

        function toggleCart() {
            const drawer = document.getElementById('cart-drawer');
            drawer.classList.toggle('hidden');
        }
